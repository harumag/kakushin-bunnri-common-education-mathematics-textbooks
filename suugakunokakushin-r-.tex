\section{指数対数関数の微分}

\begin{itembox}
[l]{\enpitu 関数の極限}
$c$を定数とする.変数$x$を定数$a$にどのような近づけ方をしても,
$|f(x)-c|$の値が限りなく0に近づくとき,
\begin{center}
$\displaystyle \lim_{x\to a}f(x)=c$
\end{center}
と表す.特に,
\begin{center}
$
\displaystyle
\lim_{x\to a}f(x)=c
\Longleftrightarrow
\begin{cases}
\displaystyle\lim_{x\to a+0}f(x)=c\\[2mm]
\displaystyle\lim_{x\to a-0}f(x)=c
\end{cases}
$
\end{center}
である.
\end{itembox}

\section{「収束速度」のはなし}

\begin{itembox}
[l]{\enpitu ランダウの記号}
\begin{align*}
\lim_{x\to a}f(x)=0,\qquad
\lim_{x\to a}g(x)=0,\qquad
\lim_{x\to a}\dfrac{f(x)}{g(x)}=0
\end{align*}
のとき,「$x=a$の近くで$f(x)$は$g(x)$より高位の無限小である」といい,
\begin{center}
$f(x)=o(g(x))\quad(x\to a)$
\end{center}
と書く.
\end{itembox}

\begin{itembox}
[l]{\enpitu 微分の定義}
関数$y=f(x)$に対し,
\begin{align*}
f'(x)
&=\lim_{\Delta x\to0}\dfrac{f(x+\Delta x)-f(x)}{\Delta x}
=\lim_{\Delta x\to0}\dfrac{\Delta y}{\Delta x}\\
&\Longleftrightarrow \lim_{\Delta x\to0}\dfrac{\Delta y-f'(x)\Delta x}{\Delta x}=0\\
&\Longleftrightarrow \Delta y-f'(x)\Delta x=o(\Delta x)\quad(\Delta x\to0)\\
&\Longleftrightarrow \Delta y=f'(x)\Delta x+o(\Delta x)\quad(\Delta x\to0)\\
&\Longleftrightarrow dy=f'(x)dx.
\end{align*}
\end{itembox}

\begin{itembox}
[l]{\enpitu 微積分学の基本定理}
$a$が定数のとき,
\begin{center}
$\displaystyle\dfrac{d}{dx}\int_a^x f(t)\,dt=f(x)$
\end{center}
である.
\end{itembox}

\section{力学系}

\begin{itembox}
[l]{\enpitu 力学系とその軌道}
漸化式
\begin{center}
$x_{n+1}=f(x_n)$
\end{center}
によって定まる数列$\{x_n\}$を,ここでは\textgt{力学系}と呼ぶ.
$x_{n+1}$は$x_n$だけから決まり,$n$が右辺のほかの位置に現れないことが特徴である.たとえば
\begin{center}
$x_{n+1}=\dfrac{1}{n}x_n(1-x_n)$
\end{center}
は,右辺に$n$が現れるので,この形の力学系ではない.

曲線$y=f(x)$と直線$y=x$を同じ座標平面に描くと,数列の動きを視覚化できる.
$x$軸上の$(x_1,0)$から曲線$y=f(x)$まで鉛直に進み,次に直線$y=x$まで水平に進む.
これを繰り返してできる折れ線を\textgt{軌道}という.
軌道を観察すると,数列の増減や,どの値へ収束するかを予想できる.

ただし,図はあくまで予想のための道具である.答案では,不等式を用いて収束を示すことが重要である.
\end{itembox}

\begin{itembox}
[l]{\enpitu 不動点と収束の基本形}
$x_n$が$\alpha$に収束するならば,漸化式$x_{n+1}=f(x_n)$において
\begin{center}
$\alpha=f(\alpha)$
\end{center}
が成り立つと考えられる.このような$\alpha$を$f$の\textgt{不動点}という.

不動点$\alpha$を見つけたら,
\begin{align*}
x_{n+1}-\alpha
&=f(x_n)-f(\alpha)
\end{align*}
と変形する.ここから,ある定数$r$について
\begin{center}
$|x_{n+1}-\alpha|\leqq r|x_n-\alpha|\qquad(0<r<1)$
\end{center}
を示すことができれば,
\begin{align*}
0\leqq|x_n-\alpha|
&\leqq r|x_{n-1}-\alpha|\\
&\leqq r^2|x_{n-2}-\alpha|\\
&\leqq\cdots\leqq r^{n-1}|x_1-\alpha|
\end{align*}
となる.$0<r<1$より$\displaystyle\lim_{n\to\infty}r^{n-1}=0$であるから,はさみうちの原理により
\begin{center}
$\displaystyle\lim_{n\to\infty}|x_n-\alpha|=0$
\end{center}
である.したがって,
\begin{center}
$\displaystyle\lim_{n\to\infty}x_n=\alpha$
\end{center}
が従う.
\end{itembox}

\begin{itembox}
[l]{\enpitu 平均値の定理による評価}
$f(x_n)-f(\alpha)$を直接変形しにくいときは,平均値の定理を用いる.
$x_n$と$\alpha$の間にある数$\beta_n$が存在して,
\begin{align*}
f(x_n)-f(\alpha)&=f'(\beta_n)(x_n-\alpha)
\end{align*}
となる.よって,考えている範囲で$|f'(x)|\leqq r<1$が成り立てば,
\begin{align*}
|x_{n+1}-\alpha|
&=|f'(\beta_n)|\,|x_n-\alpha|\\
&\leqq r|x_n-\alpha|
\end{align*}
を得る.つまり,$|f'(x)|<1$は,不動点に近づくための目安になる.
\end{itembox}

\noindent\ba{4$\bullet$1}\
数列$\{x_n\}$を
\begin{center}
$x_{n+1}=\sqrt{x_n+2}\qquad(n=1,2,3,\ldots)$
\end{center}
によって定める.$x_1$は$x_1>-2$を満たす定数とする.\
\par\noindent\kakkoichi
$\displaystyle |x_{n+1}-2|\leqq\dfrac{1}{2}|x_n-2|$が成り立つことを示せ.\
\par\noindent\kakkoni
$\displaystyle\lim_{n\to\infty}x_n$を求めよ.\\
\newpage
\noindent\hokukai\
方程式$x=\sqrt{x+2}$を解くと$x=2$となるので,不動点として$\alpha=2$を考える.\
\noindent\kakkoichib
分子を有理化すると,
\begin{align*}
x_{n+1}-2
&=\sqrt{x_n+2}-2\\
&=\dfrac{x_n-2}{\sqrt{x_n+2}+2}
\end{align*}
である.$x_n+2\geqq0$より$\sqrt{x_n+2}+2\geqq2$であるから,
\begin{align*}
|x_{n+1}-2|
&=\dfrac{|x_n-2|}{\sqrt{x_n+2}+2}\\
&\leqq\dfrac{1}{2}|x_n-2|.
\end{align*}
\noindent\kakkonib
上の不等式を繰り返し用いると,
\begin{align*}
0\leqq|x_n-2|
&\leqq\dfrac{1}{2}|x_{n-1}-2|\\
&\leqq\left(\dfrac{1}{2}\right)^2|x_{n-2}-2|\\
&\leqq\cdots\leqq\left(\dfrac{1}{2}\right)^{n-1}|x_1-2|.
\end{align*}
$\displaystyle\lim_{n\to\infty}\left(\dfrac{1}{2}\right)^{n-1}|x_1-2|=0$であるから,はさみうちの原理より
\begin{center}
$\displaystyle\ans{\lim_{n\to\infty}x_n=2}$
\end{center}
である.

\newpage

\noindent\ba{4$\bullet$2}\
$e$を自然対数の底とし,$c$を$0<c<1$を満たす定数とする.数列$\{x_n\}$を
\begin{center}
$x_1=0,\qquad x_{n+1}=ce^{-x_n}\qquad(n=1,2,3,\ldots)$
\end{center}
によって定める.\
\par\noindent\kakkoichi
方程式$x=ce^{-x}$は,$0<x<c$にただ1つの解をもつことを示せ.以下,この解を$\alpha$とする.\
\par\noindent\kakkoni
$|x_{n+1}-\alpha|\leqq c|x_n-\alpha|$が成り立つことを示し,$\displaystyle\lim_{n\to\infty}x_n$を求めよ.\\
\newpage
\noindent\hokukai\
\noindent\kakkoichib
$h(x)=x-ce^{-x}$とおくと,
\begin{align*}
h'(x)=1+ce^{-x}>0
\end{align*}
であるから,$h(x)$は単調に増加する.また,
\begin{align*}
h(0)&=-c<0,\\
h(c)&=c-ce^{-c}=c(1-e^{-c})>0
\end{align*}
である.したがって,中間値の定理と単調性より,$h(x)=0$,すなわち$x=ce^{-x}$は$0<x<c$にただ1つの解$\alpha$をもつ.\
\noindent\kakkonib
$g(x)=ce^{-x}$とおくと,$x_{n+1}=g(x_n),\ \alpha=g(\alpha)$である.
平均値の定理より,$x_n$と$\alpha$の間にある数$\beta_n$が存在して,
\begin{align*}
g(x_n)-g(\alpha)
&=g'(\beta_n)(x_n-\alpha)\\
&=-ce^{-\beta_n}(x_n-\alpha)
\end{align*}
となる.$x_n\geqq0,\ \alpha>0$より$\beta_n\geqq0$であるから,$e^{-\beta_n}\leqq1$である.よって,
\begin{align*}
|x_{n+1}-\alpha|
&=ce^{-\beta_n}|x_n-\alpha|\\
&\leqq c|x_n-\alpha|.
\end{align*}
この不等式を繰り返し用いると,
\begin{align*}
0\leqq|x_n-\alpha|
&\leqq c^{n-1}|x_1-\alpha|.
\end{align*}
$0<c<1$より$\displaystyle\lim_{n\to\infty}c^{n-1}|x_1-\alpha|=0$であるから,
\begin{center}
$\displaystyle\ans{\lim_{n\to\infty}x_n=\alpha}$
\end{center}
である.

\newpage

\noindent\ba{4$\bullet$3}\
$A$を正の定数とする.正の数列$\{x_n\}$を
\begin{center}
$x_1=A,\qquad x_{n+1}=\dfrac{1}{2}\left(x_n+\dfrac{A}{x_n}\right)\qquad(n=1,2,3,\ldots)$
\end{center}
によって定める.\
\par\noindent\kakkoichi
$n\geqq2$のとき,$x_n\geqq\sqrt{A}$が成り立つことを示せ.\
\par\noindent\kakkoni
$n\geqq2$のとき,$\displaystyle x_{n+1}-\sqrt{A}\leqq\dfrac{1}{2}(x_n-\sqrt{A})$が成り立つことを示し,$\displaystyle\lim_{n\to\infty}x_n$を求めよ.\\
\newpage
\noindent\hokukai\
\noindent\kakkoichib
漸化式の両辺から$\sqrt{A}$を引くと,
\begin{align*}
x_{n+1}-\sqrt{A}
&=\dfrac{1}{2}\left(x_n+\dfrac{A}{x_n}\right)-\sqrt{A}\\
&=\dfrac{x_n^2-2x_n\sqrt{A}+A}{2x_n}\\
&=\dfrac{(x_n-\sqrt{A})^2}{2x_n}\geqq0.
\end{align*}
したがって,$x_{n+1}\geqq\sqrt{A}$であり,$n\geqq2$のとき$x_n\geqq\sqrt{A}$である.\
\noindent\kakkonib
$n\geqq2$のとき$x_n-\sqrt{A}\geqq0$である.また,
\begin{align*}
x_{n+1}-\sqrt{A}
&=\dfrac{x_n-\sqrt{A}}{2x_n}(x_n-\sqrt{A})\\
&\leqq\dfrac{1}{2}(x_n-\sqrt{A})
\end{align*}
である.これを繰り返すと,
\begin{align*}
0\leqq x_n-\sqrt{A}
&\leqq\left(\dfrac{1}{2}\right)^{n-2}(x_2-\sqrt{A})
\end{align*}
となる.右辺は$0$に収束するから,はさみうちの原理より
\begin{center}
$\displaystyle\ans{\lim_{n\to\infty}x_n=\sqrt{A}}$
\end{center}
である.

\begin{itembox}
[l]{\enpitu この漸化式の意味}
方程式$x^2-A=0$にニュートン法を適用すると,
\begin{align*}
x_{n+1}
&=x_n-\dfrac{x_n^2-A}{2x_n}\\
&=\dfrac{1}{2}\left(x_n+\dfrac{A}{x_n}\right)
\end{align*}
となる.つまり,この漸化式は$\sqrt{A}$の近似値を高速に求める計算法である.
\end{itembox}

\section{陰関数の微分}

\begin{itembox}[l]{\enpitu 平面の交線}
$xyz$空間内の原点を通る平面
\begin{align*}
\alpha:\ z=px+qy\cdots\cdots\cdots\maruichi

\begin{itembox}[l]{\enpitu 速度ベクトルと加速度ベクトル}

時刻$t$における点Pの位置が,
\[
 \mathrm{P}\p{x(t),y(t)}
\]
によって与えられているとする.

このとき,点Pの位置ベクトルを,
\[
  \vec{r}(t)
  =
  \begin{pmatrix}
    x(t)\\
    y(t)
  \end{pmatrix}
\]
と表す.

位置ベクトル$\vec{r}(t)$を$t$で微分して得られるベクトル,

\begin{center}
  \ans{
    \vec{v}(t)
    =
    \vec{r}\,'(t)
    =
    \begin{pmatrix}
      x^\prime(t)\\
      y\prime(t)
    \end{pmatrix}
  }
\end{center}

を,時刻$t$における点Pの\ans{\text{速度ベクトル}}という.

また,速度ベクトル$\vec{v}(t)$をさらに$t$で微分して得られるベクトル,

\begin{center}
  \ans{
    \vec{a}(t)
    =
    \vec{v}\,'(t)
    =
    \vec{r}\,''(t)
    =
    \begin{pmatrix}
      x''(t)\\
      y''(t)
    \end{pmatrix}
  }
\end{center}

を,時刻$t$における点$P$の\ans{\text{加速度ベクトル}}という.
\end{itembox}

\begin{itembox}[l]{\enpitu 速度ベクトルと加速度ベクトルの関係}

速度ベクトル$\vec{v}(t)$は,
その瞬間における点$P$の進む向きを表す.

一方,加速度ベクトル$\vec{a}(t)$は,
速度ベクトルの変化する向きを表す.

したがって,加速度ベクトルが速度ベクトルに対して
左側を向いているとき,点$P$の進行方向は左向きに変化する.

同様に,加速度ベクトルが速度ベクトルに対して
右側を向いているとき,点$P$の進行方向は右向きに変化する.

  \begin{pmatrix}
    x'(t)\\

  \end{pmatrix},

\begin{aligned}
  \det\p{\vec{v}(t),\vec{a}(t)}>0
  \quad

  \vec{v}(t)\text{に対して}\vec{a}(t)\text{が左向き}

  &\narabaa\quad

  \det\p{\vec{v}(t),\vec{a}(t)}<0
  \quad
  &\narabaa\quad
  \vec{v}(t)\text{に対して}\vec{a}(t)\text{が右向き}
  \\
  &\narabaa\quad
  \text{点}P\text{は進行方向に対して右カーブ}.
\end{aligned}
\]

\begin{itembox}[l]{\enpitu 面積の増加量の表し方}

点$P(x,y)$が,
\[
  x=x(t),\qquad y=y(t)
\]

と表されているとする.

曲線と$x$軸との間の面積を求める場合,
微小な面積は縦の長さと横の変化量の積であるから,
\[
  dS=\left|y(t)x'(t)\right|\,dt
\]

と表すことができる.

同様に,曲線と$y$軸との間の面積を求める場合は,
\[
  dS=\left|x(t)y'(t)\right|\,dt
\]
となる.

また,極座標で,
\[
  r=r(t),\qquad \theta=\theta(t)
\]

と表されている場合は,微小な扇形の面積を考えることにより,
\[
  dS
  =\frac{1}{2}r(t)^2
   \left|\theta'(t)\right|\,dt
\]

となる.一応,「Gauss・Green の定理」という名前が付いています.

したがって,それぞれ,
\begin{align*}
  S
  &=
  \int_a^b
  \left|y(t)x'(t)\right|\,dt,
  \\[2mm]
  S
  &=
  \int_a^b
  \left|x(t)y'(t)\right|\,dt,
  \\[2mm]
  S
  &=
  \frac{1}{2}
  \int_a^b
  r(t)^2\left|\theta'(t)\right|\,dt
\end{align*}
と表すことができる.

\end{itembox}

$x$から$x+dx$までの部分は,幅が$dx$の非常に薄い立体である.
この立体は,底面積が$S(x)$,高さが$dx$の柱体とみなせるから,

\[
  dV\simeq S(x)\,dx
\]

となる.

したがって,$a\leq x\leq b$における立体全体の体積は,

\begin{center}
  \ans{
    V
    =
    \int_a^b dV
    =
    \int_a^b S(x)\,dx
  }
\end{center}

で求められる.

\begin{itembox}
  [l]{\enpitu 弧長の長さを求める原理}
  $t$を時刻と考えて,$t$での位置ベクトルを$\vec{r}(t)=\tvec<0,1>[x(t),y(t)]$とする.このとき,速度ベクトル$\vec{v}(t)$は,
  \begin{align*}
    \vec{v}(t)=\tvec<0,1>[\frac{d}{dt}x(t),\frac{d}{dt}y(t)]
  \end{align*}
  で表せる.ここで,微小時間$\Delta t$の間に進む距離$\Delta L(t)$は,
  \begin{align*}
    \ans{L(t)=\sqrt{\p{\frac{dx(t)}{dt}}^2+\p{\frac{dy(t)}{dt}}^2}\times\Delta t=|\vec{v}(t)|\times\Delta t}
  \end{align*}
  と表せる.これを,$t=aからt=b$まで積分すれば,弧長がでるので,全体の弧長$L(t)$は,
  \begin{align*}
    L(t)=\dint{a}{b}|\vec{v}(t)|\times dt
  \end{align*}
  となる.
\end{itembox}
